using GeometryOps as GO
using GeoInterface as GI

# Create two polygons and compute their intersection
poly1 = GI.Polygon([[(0, 0), (1, 0), (1, 1), (0, 1), (0, 0)]])
poly2 = GI.Polygon([[(0.5, 0), (1.5, 0), (1.5, 1), (0.5, 1), (0.5, 0)]])

result = GO.intersection(poly1, poly2; target = GI.PolygonTrait())
GO.area(result)  # → 0.5
